add_executable(MemLibHookTests InlineHookTests.cpp)
target_link_libraries(MemLibHookTests PRIVATE MemLib::MemLib)
if(MSVC)
  target_compile_options(MemLibHookTests PRIVATE /W4 /permissive-)
else()
  target_compile_options(MemLibHookTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
foreach(case IN ITEMS lifecycle manager relative_call relative_jump conditional internal_branch
    loop rip_relative long_instruction endbr page_boundary decode_boundary invalid_target
    duplicate target_changed move concurrent retained_original active_original calling_conventions getpc)
  add_test(NAME MemLibHook.${case} COMMAND MemLibHookTests ${case})
  set_tests_properties(MemLibHook.${case} PROPERTIES TIMEOUT 30)
endforeach()

  endforeach()
